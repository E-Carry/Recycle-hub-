Integrations

Embed the Rio widget in another site, or relay alerts through Gmail.

Web Application Integration Studio
Ready to Embed

Integrate Rio AI into any external website or web application (React, Next.js, Vue, HTML/JS) using our lightweight embed widget.

Option 1 — HTML Script Tag (Any HTML / PHP / WordPress) 

<!-- Rio AI Embed Widget -->
<script src="http://localhost:3000/rio-widget.js"
  data-name="Rio AI"
  data-primary-color="#00f2fe"
  data-position="bottom-right">
</script>

Option 2 — JavaScript ES6 / React Component Integration 

import React, { useEffect } from 'react';

export const RioAiWidget = () => {
  useEffect(() => {
    const script = document.createElement('script');
    script.src = 'http://localhost:3000/rio-widget.js';
    script.async = true;
    script.setAttribute('data-name', 'Rio AI');
    document.body.appendChild(script);
    return () => { document.body.removeChild(script); };
  }, []);
  return null;
};

Option 3 — Gmail API Relay via Google Apps Script → recycle.hub.pvt.ltd@gmail.com

Copy this script into script.google.com , click Deploy › Web App (Execute as: Me, Access: Anyone), and paste the Web App URL in Email Settings. 

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var targetEmail = "recycle.hub.pvt.ltd@gmail.com";

  MailApp.sendEmail({
    to: targetEmail,
    subject: data.subject || "[Rio AI] Problem Alert",
    body: data.body,
    htmlBody: data.htmlBody
  });

  return ContentService.createTextOutput(
    JSON.stringify({ status: "success", recipient: targetEmail })
  ).setMimeType(ContentService.MimeType.JSON);
}
